        C --> D[Configure Camera & DG645 Locally]
    end

        E[Set Filename Prefix: FileName] --> F[Set User Notes: Note]

        G --> H[Log Parameter Changes to Journal Console]
    end

        I[Call controller.arm_and_start] --> J{Interlock: Stage:IsMoving == 0?}

        L --> M[Switch Cameras to EXTERNAL_TTL Mode]
        M --> N[Set State = ARMED & Update QSS Style]
    end
    subgraph Step4 ["4. Auto-Fire & High-Speed Capture"]
        N --> O[Automatically Issue DG645 Burst: N Pulses]
        O --> P[Cameras Capture TTL Frames into Pre-allocated RAM]
        P --> Q[Update Progress & Live 2D Gaussian Preview]
    end
    subgraph Step5 ["5. Persistence, Journaling & Analysis"]
        Q --> R[Save Raw Data to HDF5/TIFF with Stage Metadata Tag]
        R --> S[Execute 2D Gaussian Fit Analysis]
        S --> T[Append Row to experiment_log.xlsx]
        T --> U[Write Run Summary to experiment_journal.log & GUI Console]
        U --> V[Reset State to IDLE & Disarm Cameras]
    end
